async function fetchRecordsFromSheets() { const statusEl = document.getElementById("syncStatus"); statusEl.innerText = "⏳ 正在從試算表讀取資料..."; statusEl.style.color = "#4f46e5"; const identifier = getUserSheetIdentifier(); const scriptUrl = `${GAS_API_URL}?email=${encodeURIComponent(identifier)}&callback=handleSheetResponse&_=${Date.now()}`; // 使用 JSONP 避開跨網域 (CORS) 擋信限制 window.handleSheetResponse = function(result) { if (result && result.status === "success") { records = result.records || []; renderList(); statusEl.innerText = "✅ 已與試算表同步"; statusEl.style.color = "#16a34a"; } else { statusEl.innerText = "⚠️ 讀取試算表失敗"; statusEl.style.color = "#dc2626"; } }; // 動態注入 script 標籤 const oldScript = document.getElementById("jsonpScript"); if (oldScript) oldScript.remove(); const script = document.createElement("script"); script.id = "jsonpScript"; script.src = scriptUrl; script.onerror = () => { // 若純 JSONP 失敗，切回一般 fetch 容錯 fetch(`${GAS_API_URL}?email=${encodeURIComponent(identifier)}`) .then(res => res.json()) .then(result => { if (result.status === "success") { records = result.records || []; renderList(); statusEl.innerText = "✅ 已與試算表同步"; statusEl.style.color = "#16a34a"; } }) .catch(() => { statusEl.innerText = "⚠️ 讀取試算表失敗"; statusEl.style.color = "#dc2626"; }); }; document.body.appendChild(script); }
